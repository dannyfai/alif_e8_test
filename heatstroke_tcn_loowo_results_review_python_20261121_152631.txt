class SeparableTemporalLayer(nn.Module):
    """Depthwise temporal convolution + pointwise channel mixing.

    Kernel 31 and dilation 1/2 are deliberate: U55 convolution hardware
    supports dilation factors 1 or 2, unlike a conventional exponentially
    dilated TCN using dilation 4..128.
    """
    def __init__(self, channels: int, kernel_size: int, dilation: int):
        super().__init__()
        effective_pad = ((kernel_size - 1) * dilation) // 2
        self.depthwise = nn.Conv1d(
            channels, channels, kernel_size=kernel_size,
            padding=effective_pad, dilation=dilation,
            groups=channels, bias=False,
        )
        self.pointwise = nn.Conv1d(channels, channels, kernel_size=1, bias=False)
        self.bn = nn.BatchNorm1d(channels)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.depthwise(x)
        x = self.pointwise(x)
        x = self.bn(x)
        return F.relu(x, inplace=False)

class TemporalResidualBlock(nn.Module):
    def __init__(self, channels: int, kernel_size: int, dilation: int):
        super().__init__()
        self.layer1 = SeparableTemporalLayer(channels, kernel_size, dilation)
        self.layer2 = SeparableTemporalLayer(channels, kernel_size, dilation)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return F.relu(x + self.layer2(self.layer1(x)), inplace=False)

class TCNRegModel(nn.Module):
    """Reg_Model-compatible replacement with only its long BiLSTM removed.

    All data preparation branches and the six per-feature projection heads are
    copied structurally from run.Reg_Model. The recurrent [B,900,16]->
    [B,900,256] stage is replaced by a residual depthwise-separable TCN.
    """
    def __init__(self, project_run, channels: int = 48, kernel_size: int = 31):
        super().__init__()
        # Instantiate the baseline to preserve exact preprocessing/projector
        # definitions and parameter names, then discard only the LSTM module.
        base = project_run.Reg_Model()
        self.seq_len = base.seq_len
        self.non_seq_len = base.non_seq_len
        self.seq_feature_num = base.seq_feature_num
        self.time_feature_num = base.time_feature_num
        self.hidden_dim1 = base.hidden_dim1
        self.dropout = base.dropout
        self.non_seq_enc = base.non_seq_enc
        self.compact_seq_len = base.compact_seq_len
        self.svf = base.svf
        self.cvf = base.cvf
        # The original six projection heads consume a 256-wide LSTM state.
        # The TCN instead predicts the six sensor channels directly with a
        # 48->6 pointwise convolution, keeping every TCN feature map <=127
        # channels for Ethos-U55 compatibility.
        del base

        input_channels = self.seq_feature_num * 2 + self.time_feature_num + 1
        if input_channels != 16:
            raise RuntimeError(f"Expected original Reg_Model embedding width 16, got {input_channels}")
        # U55 supports 1D convolution kernels up to 64 taps after dilation.
        # With dilation=2, (kernel_size - 1) * 2 + 1 must therefore be <=64.
        if channels < 1 or channels > 127:
            raise ValueError(f"Ethos-U55 TCN channels must be in [1,127], got {channels}")
        effective_dilated_kernel = (kernel_size - 1) * 2 + 1
        if kernel_size < 3 or effective_dilated_kernel > 64:
            raise ValueError(
                f"For U55 dilation=2, kernel_size must be in [3,32], got {kernel_size} "
                f"(effective kernel={effective_dilated_kernel})"
            )
        self.tcn_channels = channels
        self.tcn_kernel_size = kernel_size
        self.tcn_in = nn.Conv1d(input_channels, channels, kernel_size=1, bias=False)
        self.tcn_in_bn = nn.BatchNorm1d(channels)
        dilations = [1, 2] * 5
        self.tcn_blocks = nn.ModuleList([
            TemporalResidualBlock(channels, kernel_size, d) for d in dilations
        ])
        self.tcn_out = nn.Conv1d(channels, self.seq_feature_num, kernel_size=1, bias=True)

    def forward(self, x, x_mark_enc=None, one_hot_x=None, device=None):
        batch, length, features = x.shape
        if length != self.seq_len or features != self.seq_feature_num:
            raise ValueError(f"Expected [B,{self.seq_len},{self.seq_feature_num}], got {tuple(x.shape)}")
        device = x.device if device is None else device
        missing_mask = (x == -9997) | (x == -9998) | (x == -9999)

        if one_hot_x is None:
            raise ValueError("one_hot_x (first 18 worker features) is required")
        non_seq_enc_out = self.non_seq_enc(one_hot_x)
        if x_mark_enc is None:
            raise ValueError("x_mark_enc is required")
        time_and_worker = torch.cat((x_mark_enc, non_seq_enc_out.view(batch, -1, 1)), dim=2)

        # Original single-variate feature path.
        per_feature = []
        for b in range(batch):
            batch_features = []
            for n in range(features):
                channel = x[b:b + 1, :, n:n + 1]
                valid = (channel != -9997) & (channel != -9998) & (channel != -9999)
                values = channel[valid]
                if values.numel() > 0:
                    pooled = F.adaptive_avg_pool1d(
                        values.reshape(1, 1, -1), self.compact_seq_len
                    ).reshape(self.compact_seq_len)
                else:
                    pooled = torch.zeros(self.compact_seq_len, device=device, dtype=x.dtype)
                batch_features.append(pooled)
            per_feature.append(torch.stack(batch_features, dim=1))
        compact = torch.stack(per_feature, dim=0)
        svf_out = [layer(compact[:, :, i]) for i, layer in enumerate(self.svf)]
        svf_enc = torch.stack(svf_out, dim=2)
        svf_enc = torch.where(missing_mask, svf_enc, torch.zeros_like(svf_enc))

        # Original cross-variate feature path.
        b = x.clone()
        b[missing_mask] = float("nan")
        means = torch.nanmean(b, dim=1, keepdim=True).expand(-1, self.seq_len, -1)
        b[missing_mask] = means[missing_mask]
        b = torch.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
        cvf_enc = self.cvf(b.permute(0, 2, 1)).permute(0, 2, 1)
        cvf_enc = torch.where(missing_mask, cvf_enc, torch.zeros_like(cvf_enc))

        observed_zeroed = torch.where(missing_mask, torch.zeros_like(x), x)
        embedding = torch.cat(((svf_enc + cvf_enc) / 2.0, observed_zeroed), dim=2)
        embedding = torch.cat((embedding, time_and_worker), dim=2)

        # TCN temporal encoder. Shape: [B,16,900] -> [B,48,900].
        encoded = self.tcn_in(embedding.transpose(1, 2))
        encoded = F.relu(self.tcn_in_bn(encoded), inplace=False)
        for block in self.tcn_blocks:
            encoded = block(encoded)
        # Direct six-channel sequence reconstruction; no wide FC/projector
        # head is needed after the temporal encoder.
        return self.tcn_out(encoded).transpose(1, 2)

def load_checkpoint(path: Path) -> Dict[str, torch.Tensor]:
    try:
        obj = torch.load(path, map_location="cpu", weights_only=True)
    except TypeError:
        obj = torch.load(path, map_location="cpu")
    if isinstance(obj, dict) and "state_dict" in obj:
        obj = obj["state_dict"]
    elif isinstance(obj, dict) and "model_state_dict" in obj:
        obj = obj["model_state_dict"]
    if not isinstance(obj, dict):
        raise TypeError(f"Checkpoint is not a state_dict: {path}")
    return obj
